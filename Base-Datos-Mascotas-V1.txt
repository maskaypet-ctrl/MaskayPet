-- Mascotas QR - schema completo v1
CREATE EXTENSION IF NOT EXISTS pgcrypto;
CREATE EXTENSION IF NOT EXISTS citext;

CREATE SCHEMA IF NOT EXISTS identity;
CREATE SCHEMA IF NOT EXISTS audit;

-- IDENTITY
CREATE TABLE identity.user_accounts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email CITEXT NOT NULL UNIQUE,
    status VARCHAR(20) NOT NULL DEFAULT 'active'
        CHECK (status IN ('pending','active','suspended','deleted')),
    email_verified_at TIMESTAMPTZ,
    last_login_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE identity.user_credentials (
    user_id UUID PRIMARY KEY REFERENCES identity.user_accounts(id) ON DELETE CASCADE,
    password_hash TEXT NOT NULL,
    password_changed_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    failed_attempts INT NOT NULL DEFAULT 0 CHECK (failed_attempts >= 0),
    locked_until TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE identity.sessions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES identity.user_accounts(id) ON DELETE CASCADE,
    refresh_token_hash TEXT NOT NULL UNIQUE,
    ip_address INET,
    user_agent TEXT,
    device_name VARCHAR(100),
    expires_at TIMESTAMPTZ NOT NULL,
    revoked_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_sessions_user ON identity.sessions(user_id);
CREATE INDEX idx_sessions_expires ON identity.sessions(expires_at);

CREATE TABLE identity.password_reset_tokens (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES identity.user_accounts(id) ON DELETE CASCADE,
    token_hash TEXT NOT NULL UNIQUE,
    expires_at TIMESTAMPTZ NOT NULL,
    used_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE identity.email_verification_tokens (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES identity.user_accounts(id) ON DELETE CASCADE,
    token_hash TEXT NOT NULL UNIQUE,
    expires_at TIMESTAMPTZ NOT NULL,
    used_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- BUSINESS USER PROFILE
CREATE TABLE public.users (
    id UUID PRIMARY KEY REFERENCES identity.user_accounts(id) ON DELETE CASCADE,
    first_name VARCHAR(80) NOT NULL,
    last_name VARCHAR(80) NOT NULL,
    phone VARCHAR(30),
    avatar_storage_path TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE public.roles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    code VARCHAR(40) NOT NULL UNIQUE,
    name VARCHAR(80) NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE public.user_roles (
    user_id UUID NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
    role_id UUID NOT NULL REFERENCES public.roles(id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    PRIMARY KEY (user_id, role_id)
);

-- PLANS
CREATE TABLE public.catalog_tiers (
    tier VARCHAR(20) PRIMARY KEY CHECK (tier IN ('free','premium'))
);
INSERT INTO public.catalog_tiers(tier) VALUES ('free'),('premium')
ON CONFLICT DO NOTHING;

CREATE TABLE public.plan_limits (
    tier VARCHAR(20) PRIMARY KEY REFERENCES public.catalog_tiers(tier),
    max_pets INT CHECK (max_pets IS NULL OR max_pets > 0),
    max_collaborators_per_pet INT NOT NULL CHECK (max_collaborators_per_pet >= 0),
    sms_credits_monthly INT NOT NULL DEFAULT 0 CHECK (sms_credits_monthly >= 0),
    medical_document_retention_years INT NOT NULL DEFAULT 2 CHECK (medical_document_retention_years > 0)
);

INSERT INTO public.plan_limits VALUES
('free',1,1,0,1),
('premium',NULL,5,10,10)
ON CONFLICT (tier) DO UPDATE SET
max_pets=EXCLUDED.max_pets,
max_collaborators_per_pet=EXCLUDED.max_collaborators_per_pet,
sms_credits_monthly=EXCLUDED.sms_credits_monthly,
medical_document_retention_years=EXCLUDED.medical_document_retention_years;

-- PETS
CREATE TABLE public.pets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    owner_id UUID NOT NULL REFERENCES public.users(id) ON DELETE RESTRICT,
    name VARCHAR(100) NOT NULL,
    species VARCHAR(30) NOT NULL,
    breed VARCHAR(100),
    sex VARCHAR(20) CHECK (sex IN ('male','female','unknown')),
    birth_date DATE,
    color VARCHAR(100),
    microchip_number VARCHAR(50) UNIQUE,
    is_sterilized BOOLEAN,
    description TEXT,
    photo_storage_path TEXT,
    status VARCHAR(20) NOT NULL DEFAULT 'active'
      CHECK (status IN ('active','deceased','transferred','archived')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_pets_owner ON public.pets(owner_id);

CREATE TABLE public.pet_contacts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    name VARCHAR(120) NOT NULL,
    relationship VARCHAR(50),
    phone VARCHAR(30),
    email CITEXT,
    whatsapp VARCHAR(30),
    priority INT NOT NULL DEFAULT 1 CHECK (priority > 0),
    is_primary BOOLEAN NOT NULL DEFAULT FALSE,
    can_receive_lost_alerts BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_pet_contacts_pet ON public.pet_contacts(pet_id);

CREATE TABLE public.pet_collaborators (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
    role VARCHAR(20) NOT NULL DEFAULT 'viewer'
      CHECK (role IN ('caregiver','viewer')),
    invited_by UUID REFERENCES public.users(id) ON DELETE SET NULL,
    invited_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    accepted_at TIMESTAMPTZ,
    UNIQUE (pet_id,user_id)
);
CREATE INDEX idx_collaborators_pet ON public.pet_collaborators(pet_id);
CREATE INDEX idx_collaborators_user ON public.pet_collaborators(user_id);

CREATE TABLE public.pet_public_profiles (
    pet_id UUID PRIMARY KEY REFERENCES public.pets(id) ON DELETE CASCADE,
    show_pet_name BOOLEAN NOT NULL DEFAULT TRUE,
    show_photo BOOLEAN NOT NULL DEFAULT TRUE,
    show_breed BOOLEAN NOT NULL DEFAULT TRUE,
    show_owner_name BOOLEAN NOT NULL DEFAULT FALSE,
    show_owner_phone BOOLEAN NOT NULL DEFAULT FALSE,
    show_contacts BOOLEAN NOT NULL DEFAULT FALSE,
    show_medical_info BOOLEAN NOT NULL DEFAULT FALSE,
    emergency_message TEXT,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- HEALTH
CREATE TABLE public.pet_vaccinations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    vaccine_name VARCHAR(150) NOT NULL,
    application_date DATE NOT NULL,
    next_due_date DATE,
    batch_number VARCHAR(80),
    veterinarian_id UUID REFERENCES public.veterinarians(id) ON DELETE SET NULL,
    clinic_id UUID REFERENCES public.clinics(id) ON DELETE SET NULL,
    veterinarian_name VARCHAR(120),
    clinic_name VARCHAR(150),
    notes TEXT,
    created_by UUID REFERENCES public.users(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    CHECK (next_due_date IS NULL OR next_due_date >= application_date)
);
CREATE INDEX idx_vaccinations_pet_date ON public.pet_vaccinations(pet_id, application_date DESC);
CREATE INDEX idx_vaccinations_due ON public.pet_vaccinations(next_due_date);

CREATE TABLE public.pet_deworming (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    product_name VARCHAR(100) NOT NULL,
    application_date DATE NOT NULL,
    next_due_date DATE,
    weight_at_application_kg NUMERIC(6,2) CHECK (weight_at_application_kg IS NULL OR weight_at_application_kg > 0),
    veterinarian_id UUID REFERENCES public.veterinarians(id) ON DELETE SET NULL,
    clinic_id UUID REFERENCES public.clinics(id) ON DELETE SET NULL,
    notes VARCHAR(250),
    created_by UUID REFERENCES public.users(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    CHECK (next_due_date IS NULL OR next_due_date >= application_date)
);
CREATE INDEX idx_deworming_pet ON public.pet_deworming(pet_id);
CREATE INDEX idx_deworming_due ON public.pet_deworming(next_due_date);

CREATE TABLE public.pet_treatments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    treatment_name VARCHAR(150) NOT NULL,
    medication_name VARCHAR(150),
    dosage VARCHAR(100),
    frequency VARCHAR(100),
    start_date DATE NOT NULL,
    end_date DATE,
    instructions TEXT,
    veterinarian_id UUID REFERENCES public.veterinarians(id) ON DELETE SET NULL,
    veterinarian_name VARCHAR(120),
    status VARCHAR(20) NOT NULL DEFAULT 'active'
      CHECK (status IN ('planned','active','completed','canceled')),
    created_by UUID REFERENCES public.users(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    CHECK (end_date IS NULL OR end_date >= start_date)
);
CREATE INDEX idx_treatments_pet ON public.pet_treatments(pet_id);
CREATE INDEX idx_treatments_status ON public.pet_treatments(pet_id,status);

CREATE TABLE public.pet_diagnoses (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    diagnosis VARCHAR(200) NOT NULL,
    diagnosis_date DATE NOT NULL,
    description TEXT,
    veterinarian_id UUID REFERENCES public.veterinarians(id) ON DELETE SET NULL,
    clinic_id UUID REFERENCES public.clinics(id) ON DELETE SET NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'active'
      CHECK (status IN ('active','resolved','chronic')),
    created_by UUID REFERENCES public.users(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_diagnoses_pet_date ON public.pet_diagnoses(pet_id,diagnosis_date DESC);

CREATE TABLE public.pet_weight_records (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    weight_kg NUMERIC(6,2) NOT NULL CHECK (weight_kg > 0),
    measured_at DATE NOT NULL,
    notes VARCHAR(250),
    created_by UUID REFERENCES public.users(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_weight_pet_date ON public.pet_weight_records(pet_id,measured_at DESC);

CREATE TABLE public.appointments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    user_id UUID REFERENCES public.users(id) ON DELETE SET NULL,
    veterinarian_id UUID REFERENCES public.veterinarians(id) ON DELETE SET NULL,
    clinic_id UUID REFERENCES public.clinics(id) ON DELETE SET NULL,
    scheduled_at TIMESTAMPTZ NOT NULL,
    reason VARCHAR(200),
    notes TEXT,
    status VARCHAR(20) NOT NULL DEFAULT 'scheduled'
      CHECK (status IN ('scheduled','confirmed','completed','canceled','no_show')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_appointments_pet_date ON public.appointments(pet_id,scheduled_at);

CREATE TABLE public.pet_medical_documents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    uploaded_by UUID REFERENCES public.users(id) ON DELETE SET NULL,
    document_type VARCHAR(30) NOT NULL
      CHECK (document_type IN ('prescription','lab_result','diagnosis','medical_report','vaccination_card','other')),
    file_name VARCHAR(255) NOT NULL,
    storage_path TEXT NOT NULL UNIQUE,
    mime_type VARCHAR(100),
    file_size_bytes BIGINT CHECK (file_size_bytes IS NULL OR file_size_bytes >= 0),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_medical_docs_pet ON public.pet_medical_documents(pet_id);

-- QR
CREATE TABLE public.tag_production_batches (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    batch_code VARCHAR(50) NOT NULL UNIQUE,
    quantity INT NOT NULL CHECK (quantity > 0),
    produced_at TIMESTAMPTZ,
    created_by UUID REFERENCES public.users(id) ON DELETE SET NULL,
    notes TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE public.qr_tags (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    production_batch_id UUID REFERENCES public.tag_production_batches(id) ON DELETE SET NULL,
    public_code VARCHAR(32) NOT NULL UNIQUE,
    status VARCHAR(20) NOT NULL DEFAULT 'available'
      CHECK (status IN ('available','active','inactive','lost','damaged','retired')),
    printed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_qr_tags_status ON public.qr_tags(status);

CREATE TABLE public.qr_tag_assignments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    qr_tag_id UUID NOT NULL REFERENCES public.qr_tags(id) ON DELETE CASCADE,
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE RESTRICT,
    assigned_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    unassigned_at TIMESTAMPTZ,
    reason VARCHAR(100),
    CHECK (unassigned_at IS NULL OR unassigned_at >= assigned_at)
);
CREATE INDEX idx_tag_assignments_tag ON public.qr_tag_assignments(qr_tag_id);
CREATE INDEX idx_tag_assignments_pet ON public.qr_tag_assignments(pet_id);

-- EMERGENCY
CREATE TABLE public.lost_mode_events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pet_id UUID NOT NULL REFERENCES public.pets(id) ON DELETE CASCADE,
    activated_by UUID REFERENCES public.users(id) ON DELETE SET NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'active'
      CHECK (status IN ('active','recovered','canceled','expired')),
    activated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    resolved_at TIMESTAMPTZ,
    recovery_method VARCHAR(30)
      CHECK (recovery_method IS NULL OR recovery_method IN ('via_scan','otro_medio')),
    resolution_ping_id UUID,
    owner_testimonial TEXT
);
CREATE INDEX idx_lost_pet_status ON public.lost_mode_events(pet_id,status);

CREATE TABLE public.profile_scan_events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    qr_tag_id UUID NOT NULL REFERENCES public.qr_tags(id) ON DELETE CASCADE,
    pet_id UUID REFERENCES public.pets(id) ON DELETE SET NULL,
    was_lost_mode BOOLEAN NOT NULL DEFAULT FALSE,
    device_type VARCHAR(20)
      CHECK (device_type IS NULL OR device_type IN ('ios','android','desktop','other')),
    scanned_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_scan_events_tag_date ON public.profile_scan_events(qr_tag_id,scanned_at DESC);
CREATE INDEX idx_scan_events_pet_date ON public.profile_scan_events(pet_id,scanned_at DESC);

CREATE TABLE public.location_pings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    lost_mode_event_id UUID REFERENCES public.lost_mode_events(id) ON DELETE CASCADE,
    qr_tag_id UUID REFERENCES public.qr_tags(id) ON DELETE SET NULL,
    latitude NUMERIC(9,6) NOT NULL CHECK (latitude BETWEEN -90 AND 90),
    longitude NUMERIC(9,6) NOT NULL CHECK (longitude BETWEEN -180 AND 180),
    accuracy_m NUMERIC(8,2) CHECK (accuracy_m IS NULL OR accuracy_m >= 0),
    captured_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    purge_after TIMESTAMPTZ NOT NULL DEFAULT (now() + INTERVAL '90 days')
);
CREATE INDEX idx_location_pings_event_date ON public.location_pings(lost_mode_event_id,captured_at DESC);
CREATE INDEX idx_location_pings_purge ON public.location_pings(purge_after);

ALTER TABLE public.lost_mode_events
ADD CONSTRAINT fk_lost_resolution_ping
FOREIGN KEY (resolution_ping_id) REFERENCES public.location_pings(id) ON DELETE SET NULL;

-- SUBSCRIPTIONS / BILLING
CREATE TABLE public.subscriptions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES public.users(id) ON DELETE RESTRICT,
    tier VARCHAR(20) NOT NULL REFERENCES public.catalog_tiers(tier),
    status VARCHAR(20) NOT NULL DEFAULT 'active'
      CHECK (status IN ('trialing','active','past_due','canceled','expired')),
    started_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    current_period_start TIMESTAMPTZ,
    current_period_end TIMESTAMPTZ,
    canceled_at TIMESTAMPTZ,
    auto_renew BOOLEAN NOT NULL DEFAULT TRUE,
    provider VARCHAR(30),
    provider_subscription_id VARCHAR(150),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_subscriptions_user_status ON public.subscriptions(user_id,status);

CREATE TABLE public.sms_credit_usage (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
    period_start DATE NOT NULL,
    period_end DATE NOT NULL,
    credits_used INT NOT NULL DEFAULT 0 CHECK (credits_used >= 0),
    credits_limit INT NOT NULL CHECK (credits_limit >= 0),
    UNIQUE (user_id,period_start),
    CHECK (period_end >= period_start)
);

CREATE TABLE public.payments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.users(id) ON DELETE SET NULL,
    subscription_id UUID REFERENCES public.subscriptions(id) ON DELETE SET NULL,
    amount NUMERIC(12,2) NOT NULL CHECK (amount >= 0),
    currency CHAR(3) NOT NULL DEFAULT 'PEN',
    status VARCHAR(20) NOT NULL DEFAULT 'pending'
      CHECK (status IN ('pending','approved','rejected','refunded','canceled')),
    provider VARCHAR(30),
    provider_payment_id VARCHAR(150),
    paid_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_payments_user_date ON public.payments(user_id,created_at DESC);

-- COMMERCE
CREATE TABLE public.products (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    description TEXT,
    sku VARCHAR(50) UNIQUE,
    price NUMERIC(10,2) NOT NULL CHECK (price >= 0),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE public.orders (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES public.users(id) ON DELETE RESTRICT,
    payment_id UUID REFERENCES public.payments(id) ON DELETE SET NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'pending'
      CHECK (status IN ('pending','paid','processing','shipped','delivered','canceled')),
    payment_status VARCHAR(20) NOT NULL DEFAULT 'pending'
      CHECK (payment_status IN ('pending','approved','rejected','refunded')),
    shipping_address VARCHAR(250) NOT NULL,
    shipping_city VARCHAR(80) NOT NULL,
    shipping_country VARCHAR(60) NOT NULL DEFAULT 'Perú',
    tracking_number VARCHAR(60),
    subtotal NUMERIC(10,2) NOT NULL DEFAULT 0 CHECK (subtotal >= 0),
    discount NUMERIC(10,2) NOT NULL DEFAULT 0 CHECK (discount >= 0),
    shipping_cost NUMERIC(10,2) NOT NULL DEFAULT 0 CHECK (shipping_cost >= 0),
    total NUMERIC(10,2) NOT NULL DEFAULT 0 CHECK (total >= 0),
    currency CHAR(3) NOT NULL DEFAULT 'PEN',
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_orders_user_date ON public.orders(user_id,created_at DESC);

CREATE TABLE public.order_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    order_id UUID NOT NULL REFERENCES public.orders(id) ON DELETE CASCADE,
    product_id UUID REFERENCES public.products(id) ON DELETE SET NULL,
    qr_tag_id UUID REFERENCES public.qr_tags(id) ON DELETE SET NULL,
    product_name VARCHAR(100) NOT NULL,
    quantity INT NOT NULL DEFAULT 1 CHECK (quantity > 0),
    unit_price NUMERIC(10,2) NOT NULL CHECK (unit_price >= 0),
    total_price NUMERIC(10,2) GENERATED ALWAYS AS (quantity * unit_price) STORED
);
CREATE INDEX idx_order_items_order ON public.order_items(order_id);

-- MARKETING
CREATE TABLE public.coupons (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    code VARCHAR(30) NOT NULL UNIQUE,
    discount_type VARCHAR(10) NOT NULL CHECK (discount_type IN ('percent','fixed')),
    discount_value NUMERIC(10,2) NOT NULL CHECK (discount_value >= 0),
    referrer_user_id UUID REFERENCES public.users(id) ON DELETE SET NULL,
    valid_from DATE NOT NULL DEFAULT CURRENT_DATE,
    valid_until DATE,
    max_uses INT CHECK (max_uses IS NULL OR max_uses > 0),
    times_used INT NOT NULL DEFAULT 0 CHECK (times_used >= 0),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    CHECK (valid_until IS NULL OR valid_until >= valid_from),
    CHECK (discount_type <> 'percent' OR discount_value <= 100)
);

CREATE TABLE audit.audit_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    schema_name VARCHAR(63) NOT NULL,
    table_name VARCHAR(63) NOT NULL,
    record_id UUID,
    action VARCHAR(10) NOT NULL CHECK (action IN ('INSERT','UPDATE','DELETE')),
    old_data JSONB,
    new_data JSONB,
    changed_by UUID,
    changed_at TIMESTAMPTZ NOT NULL DEFAULT now(),

);

CREATE INDEX idx_audit_user ON audit.audit_logs(changed_by,changed_at DESC);

-- trigger updated_at
CREATE OR REPLACE FUNCTION public.set_updated_at()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    NEW.updated_at = now();
    RETURN NEW;
END $$;

DO $$
DECLARE t TEXT;
BEGIN
    FOREACH t IN ARRAY ARRAY[
        'identity.user_accounts','identity.user_credentials','public.users',
        'public.pets','public.pet_contacts','public.pet_treatments',
        'public.pet_diagnoses','public.pet_public_profiles','public.clinics',
        'public.appointments','public.notification_preferences',
        'public.subscriptions','public.products','public.orders'
    ]
    LOOP
        EXECUTE format('DROP TRIGGER IF EXISTS trg_updated_at ON %s', t);
        EXECUTE format('CREATE TRIGGER trg_updated_at BEFORE UPDATE ON %s FOR EACH ROW EXECUTE FUNCTION public.set_updated_at()', t);
    END LOOP;
END $$;

-- Basic function to record audit entries; app should set app.user_id/app.ip_address.
CREATE OR REPLACE FUNCTION audit.fn_audit_trigger()
RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
    actor UUID;
    ip INET;
BEGIN
    BEGIN actor := NULLIF(current_setting('app.user_id', true),'')::UUID; EXCEPTION WHEN OTHERS THEN actor := NULL; END;
    BEGIN ip := NULLIF(current_setting('app.ip_address', true),'')::INET; EXCEPTION WHEN OTHERS THEN ip := NULL; END;

    IF TG_OP = 'INSERT' THEN
        INSERT INTO audit.audit_logs(schema_name,table_name,record_id,action,new_data,changed_by,ip_address)
        VALUES (TG_TABLE_SCHEMA,TG_TABLE_NAME,NEW.id,'INSERT',to_jsonb(NEW),actor,ip);
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO audit.audit_logs(schema_name,table_name,record_id,action,old_data,new_data,changed_by,ip_address)
        VALUES (TG_TABLE_SCHEMA,TG_TABLE_NAME,NEW.id,'UPDATE',to_jsonb(OLD),to_jsonb(NEW),actor,ip);
        RETURN NEW;
    ELSE
        INSERT INTO audit.audit_logs(schema_name,table_name,record_id,action,old_data,changed_by,ip_address)
        VALUES (TG_TABLE_SCHEMA,TG_TABLE_NAME,OLD.id,'DELETE',to_jsonb(OLD),actor,ip);
        RETURN OLD;
    END IF;
END $$;

-- Audit medical and security-sensitive business tables.
DO $$
DECLARE t TEXT;
BEGIN
    FOREACH t IN ARRAY ARRAY[
        'public.pets','public.pet_vaccinations','public.pet_deworming',
        'public.pet_treatments','public.pet_diagnoses','public.pet_weight_records',
        'public.pet_medical_documents','public.appointments',
        'public.pet_collaborators','public.pet_contacts','public.pet_public_profiles',
        'public.lost_mode_events','public.qr_tag_assignments','public.subscriptions'
    ]
    LOOP
        EXECUTE format('DROP TRIGGER IF EXISTS trg_audit ON %s', t);
        EXECUTE format('CREATE TRIGGER trg_audit AFTER INSERT OR UPDATE OR DELETE ON %s FOR EACH ROW EXECUTE FUNCTION audit.fn_audit_trigger()', t);
    END LOOP;
END $$;
